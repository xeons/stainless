# The class and its metaclass, in __objc_data. The metaclass's isa and its
# superclass are both NSObject's metaclass, NSObject being the root.
# clang: @"OBJC_METACLASS_$_Canvas" = global %struct._class_t { ptr @"OBJC_METACLASS_$_NSObject",
#        ptr @"OBJC_METACLASS_$_NSObject", ptr @_objc_empty_cache, ptr null, ptr @"_OBJC_METACLASS_RO_$_Canvas" },
#        section "__DATA, __objc_data", align 8
@"OBJC_METACLASS_$_ObjCClass.Canvas" = global { ptr, ptr, ptr, ptr, ptr } { ptr @"OBJC_METACLASS_$_NSObject", ptr @"OBJC_METACLASS_$_NSObject", ptr @_objc_empty_cache, ptr null, ptr @"_OBJC_METACLASS_RO_$_ObjCClass.Canvas" }, section "__DATA, __objc_data", align 8

# class_ro_t. RO_HAS_CXX_STRUCTORS (4) for .cxx_construct and .cxx_destruct,
# and RO_META (1) on the metaclass, which is 40 bytes. clang also sets
# RO_IS_ARC (0x80); the runtime has no part in what the ivar holds here, so
# it is left off. The instance starts after the isa and the runtime slides it.
# clang: @"_OBJC_CLASS_RO_$_Canvas" = internal global %struct._class_ro_t { i32 388, i32 8, i32 24, ...
@"_OBJC_METACLASS_RO_$_ObjCClass.Canvas" = internal global { i32, i32, i32, ptr, ptr, ptr, ptr, ptr, ptr, ptr } { i32 5, i32 40, i32 40,

section "__DATA, __objc_const", align 8
