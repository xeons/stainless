# The class against what clang writes for the same class in Objective-C, with
# --target=x86_64-apple-macosx13.0 -fobjc-arc -O0. What differs from arm64 is
# what this pins: BOOL, a struct returned in memory, and the ivar offset.

# BOOL is a signed char on Intel, in the encoding and in the IMP.
# clang: c"{CGRect={CGPoint=dd}{CGSize=dd}}20@0:8c16\00"
#        define internal void @"\01-[E big:]"(ptr ... sret(%struct.Big) align 8 %0, ptr noundef %1,
#        ptr noundef %2, i8 noundef signext %3)
c"c20@0:8c16\00", section "__TEXT,__objc_methtype,cstring_literals"
define internal signext i8 @"\01-[ObjCClass.Canvas isAbove:]"(ptr %self, ptr %_cmd, i8 signext %arg.flag)
icmp ne i8 %arg.flag, 0
zext i1

# A struct returned in memory: the hidden pointer comes first, before self.
define internal void @"\01-[ObjCClass.Canvas spread]"(ptr sret(%struct.ObjCClass_Wide) %sret, ptr %self, ptr %_cmd)
c"{Wide=dddd}16@0:8\00", section "__TEXT,__objc_methtype,cstring_literals"

# The ivar offset is a word on Intel.
# clang: @"OBJC_IVAR_$_Canvas._count" = hidden constant i64 8, section "__DATA, __objc_ivar", align 8
@"OBJC_IVAR_$_ObjCClass.Canvas._sl" = hidden global i64 8, section "__DATA, __objc_ivar", align 8
load i64, ptr @"OBJC_IVAR_$_ObjCClass.Canvas._sl", align 8

# The rest is the same layout as arm64.
@"OBJC_CLASS_$_ObjCClass.Canvas" = global { ptr, ptr, ptr, ptr, ptr } { ptr @"OBJC_METACLASS_$_ObjCClass.Canvas", ptr @"OBJC_CLASS_$_NSObject", ptr @_objc_empty_cache, ptr null, ptr @"_OBJC_CLASS_RO_$_ObjCClass.Canvas" }, section "__DATA, __objc_data", align 8
@"_OBJC_CLASS_RO_$_ObjCClass.Canvas" = internal global { i32, i32, i32, ptr, ptr, ptr, ptr, ptr, ptr, ptr } { i32 4, i32 8, i32 24,
@"OBJC_LABEL_CLASS_$" = private global [1 x ptr] [ptr @"OBJC_CLASS_$_ObjCClass.Canvas"], section "__DATA,__objc_classlist,regular,no_dead_strip", align 8
